598constexpr Error Validate(const Config& config) noexcept {
599 const bool sdo_ok =
600 (config.sdo_timeout_us >= 1'000U) && (config.sdo_timeout_us <= kMaxSdoTimeoutUs);
601 const bool guard_ok =
602 (config.guard_timeout_us >= 1'000U) && (config.guard_timeout_us <= 10'000'000U);
603 const bool drive_ok =
604 (config.drive_timeout_us >= 1'000U) && (config.drive_timeout_us <= kMaxSdoTimeoutUs);
605 const bool poll_ok =
606 (config.drive_poll_interval_us >= 100U) && (config.drive_poll_interval_us <= 1'000'000U);
607 const bool attempts_ok = (config.safety_attempts >= 1U) && (config.safety_attempts <= 8U);
608 if (sdo_ok && guard_ok && drive_ok && poll_ok && attempts_ok) {
609 return Error::Ok;
610 }
611 return Error::InvalidArgument;
612}
