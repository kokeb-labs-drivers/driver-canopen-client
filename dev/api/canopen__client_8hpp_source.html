31// Optional hook: a `Declares*` concept only tests the name, the `Has*` concept the exact signature,
32// `noexcept` and the result type; the driver rejects a hook that is declared but does not match
33// (handbook KB-CPP20-007).
34
36template <typename Port>
37concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };
38
40template <typename Port>
41concept HasLog =
42 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };
43
45template <typename Port>
46concept DeclaresReceive = requires(Port& port, CanFrame& frame) { port.Receive(frame, 0_u32); };
47
49template <typename Port>
50concept DeclaresNowUs = requires(Port& port) { port.NowUs(); };
51
53template <typename Port>
54void Log(Port& port, [[maybe_unused]] LogLevel level,
55 [[maybe_unused]] const char* message) noexcept {
56 if constexpr (HasLog<Port>) {
57 port.Log(level, message);
58 } else {
59 static_cast<void>(port);
60 }
61}
