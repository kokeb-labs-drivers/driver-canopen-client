336constexpr Result<SdoReply> ParseInitiateUpload(const CanFrame& f, SdoReply r,
337 bool strict) noexcept {
338 using detail::operator""_u8;
339 const uint32_t b0 = Widen(f.data[0]);
340 const bool expedited = TestBit(b0, 1U);
341 const bool sized = TestBit(b0, 0U);
342 r.expedited = expedited;
343 r.size_indicated = sized;
344 if (expedited) {
345 const uint32_t len = sized ? (4U - ((b0 >> 2U) & 3U)) : 4U;
346 if ((Widen(f.dlc) < (4U + len)) || (strict && (f.dlc != 8_u8))) {
347 return Result<SdoReply>::Failure(Error::InvalidFrame);
348 }
349 r.data_len = static_cast<uint8_t>(len);
350 for (uint32_t i = 0U; i < len; ++i) {
351 r.data[i] = f.data[4U + i];
352 }
353 return Result<SdoReply>::Success(r);
354 }
355 const uint32_t need = sized ? 8U : 4U;
356 if ((Widen(f.dlc) < need) || (strict && (f.dlc != 8_u8))) {
357 return Result<SdoReply>::Failure(Error::InvalidFrame);
358 }
359 r.size = sized ? LoadLe32At4(f.data) : 0U;
360 return Result<SdoReply>::Success(r);
361}

415constexpr Result<SdoReply> ParseSdoReply(const CanFrame& frame, bool strict) noexcept {
416 using detail::operator""_u8;
417 if (frame.rtr || (frame.dlc < 1_u8) || (frame.dlc > kMaxDlc)) {
418 return Result<SdoReply>::Failure(Error::InvalidFrame);
419 }
420 SdoReply r{};
421 const uint32_t b0 = detail::Widen(frame.data[0]);
422 const uint32_t scs = b0 >> 5U;
423 const bool header_ok = (frame.dlc >= 4_u8);
424 if (scs == 1U) { // download segment response
425 return detail::ParseDownloadSegment(frame, r, strict);
426 }
427 if (scs == 0U) { // upload segment
428 r.kind = SdoReplyKind::SegmentUpload;
429 return detail::ParseUploadSegment(frame, r, strict);
430 }
431 if (!header_ok || ((scs != 4U) && strict && (frame.dlc != 8_u8))) {
432 return Result<SdoReply>::Failure(Error::InvalidFrame);
433 }
434 r.index = detail::Le16(frame.data[1], frame.data[2]);
435 r.sub = frame.data[3];
436 if (scs == 3U) { // initiate download response
437 r.kind = SdoReplyKind::InitiateDownload;
438 return Result<SdoReply>::Success(r);
439 }
440 if (scs == 2U) { // initiate upload response
441 r.kind = SdoReplyKind::InitiateUpload;
442 return detail::ParseInitiateUpload(frame, r, strict);
443 }
444 return detail::ParseAbort(frame, r, scs);
445}

453constexpr Result<HeartbeatInfo> ParseHeartbeat(const CanFrame& frame) noexcept {
454 using detail::operator""_u8;
455 const uint32_t id = detail::Widen(frame.id);
456 const uint32_t base = detail::Widen(kCobIdHeartbeatBase);
457 if (frame.rtr || (frame.dlc != 1_u8) || (id <= base) ||
458 (id > (base + detail::Widen(kMaxNodeId)))) {
459 return Result<HeartbeatInfo>::Failure(Error::InvalidFrame);
460 }
461 HeartbeatInfo h{};
462 h.node = static_cast<uint8_t>(id - base);
463 h.toggle = detail::TestBit(detail::Widen(frame.data[0]), 7U);
464 const auto raw = static_cast<uint8_t>(detail::Widen(frame.data[0]) & 0x7FU);
465 if (!DecodeNmtState(raw, h.state)) {
466 return Result<HeartbeatInfo>::Failure(Error::InvalidFrame);
467 }
468 return Result<HeartbeatInfo>::Success(h);
469}

556constexpr Error ValidatePdoConfig(const PdoConfig& config) noexcept {
557 using detail::operator""_u8;
558 using detail::operator""_u16;
559 if ((config.cob_id < kPdoCobIdMin) || (config.cob_id > kPdoCobIdMax) ||
560 (config.entry_count > kMaxPdoEntries)) {
561 return Error::InvalidArgument;
562 }
563 if ((config.transmission_type >= 241_u8) && (config.transmission_type <= 251_u8)) {
564 return Error::InvalidArgument; // reserved
565 }
566 uint32_t total = 0U;
567 for (std::size_t i = 0; i < config.entry_count; ++i) {
568 const PdoMapEntry& e = config.entries[i];
569 if ((e.index == 0_u16) || (e.bits < 1_u8) || (e.bits > 64_u8)) {
570 return Error::InvalidArgument;
571 }
572 total += detail::Widen(e.bits);
573 }
574 return (total <= 64U) ? Error::Ok : Error::InvalidArgument;
575}

599constexpr Result<uint64_t> ExtractBits(std::span<const uint8_t> data, uint32_t offset,
600 uint32_t bits) noexcept {
601 const uint64_t end = static_cast<uint64_t>(offset) + bits;
602 if ((bits < 1U) || (bits > 64U) || (end > (static_cast<uint64_t>(data.size()) * 8U))) {
603 return Result<uint64_t>::Failure(Error::InvalidArgument);
604 }
605 uint64_t value = 0U;
606 for (uint32_t i = 0U; i < bits; ++i) {
607 const uint32_t bit = offset + i;
608 const uint32_t b = (detail::Widen(data[bit / 8U]) >> (bit % 8U)) & 1U;
609 value |= static_cast<uint64_t>(b) << i;
610 }
611 return Result<uint64_t>::Success(value);
612}

623constexpr Result<void> InsertBits(std::span<uint8_t> data, uint32_t offset, uint32_t bits,
624 uint64_t value) noexcept {
625 const uint64_t end = static_cast<uint64_t>(offset) + bits;
626 if ((bits < 1U) || (bits > 64U) || (end > (static_cast<uint64_t>(data.size()) * 8U))) {
627 return Result<void>::Failure(Error::InvalidArgument);
628 }
629 for (uint32_t i = 0U; i < bits; ++i) {
630 const uint32_t bit = offset + i;
631 const uint32_t mask = 1U << (bit % 8U);
632 const uint32_t old = detail::Widen(data[bit / 8U]);
633 const bool set = ((value >> i) & 1U) != 0U;
634 data[bit / 8U] = static_cast<uint8_t>(set ? (old | mask) : (old & ~mask));
635 }
636 return Result<void>::Success();
637}

647constexpr DriveState DecodeStatusword(uint16_t statusword) noexcept {
648 const uint32_t sw = detail::Widen(statusword);
649 const uint32_t low = sw & 0x4FU;
650 const uint32_t quick = sw & 0x6FU;
651 if (low == 0x00U) {
652 return DriveState::NotReadyToSwitchOn;
653 }
654 if (low == 0x40U) {
655 return DriveState::SwitchOnDisabled;
656 }
657 if (low == 0x0FU) {
658 return DriveState::FaultReactionActive;
659 }
660 if (low == 0x08U) {
661 return DriveState::Fault;
662 }
663 if (quick == 0x21U) {
664 return DriveState::ReadyToSwitchOn;
665 }
666 if (quick == 0x23U) {
667 return DriveState::SwitchedOn;
668 }
669 if (quick == 0x27U) {
670 return DriveState::OperationEnabled;
671 }
672 if (quick == 0x07U) {
673 return DriveState::QuickStopActive;
674 }
675 return DriveState::Unknown;
676}
